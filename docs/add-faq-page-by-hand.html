APEX Feedback Hub · Add-on to an installed Hub · Plan P68

Plan P68 · FAQ for IT

Adds the FAQ page to a Hub installed before Release 1.0.60, without re-importing the application. If Administration already shows a FAQ for IT card, it is installed; stop here. Sequenced: the file, then the page, then the Administration card, then the menu entry. Click any value to copy.

Steps
  1. #1Upload the FAQ file
  2. #2Create page 68
  3. #3Add the FAQ card to Administration
  4. #4Add the menu entry
#1

Upload the FAQ as a static application file

Steps

  1. Take docs/afh_faq.html from the release bundle this plan came in (v0.1.1-r1.0.62 or later, https://github.com/praveensehgal/apex-feedback-hub/releases). It is plain ASCII.
  2. In the Hub application: Shared Components → Static Application Files → Create File.
  3. Upload the file. Keep the file name exactly afh_faq.html and no directory: the page looks the file up by that name, and shows "The FAQ file afh_faq.html is missing" if it is renamed.
  4. MIME type: text/html. Character set: UTF-8. Create.
#2

Create page 68, FAQ for IT

Steps

  1. Create Page → Blank Page. Page Number 68, Name FAQ for IT, Page Mode Normal.
  2. Breadcrumb: do not use. Navigation: turn Use Navigation off (the Hub builds its menu from the database; step #4 adds the entry). Create Page.
  3. If the wizard created a Breadcrumb region, delete it.
  4. Page attributes, set as in the table below.
  5. Page → JavaScript → Execute when Page Loads: paste Page load JavaScript.
  6. Page → CSS → Inline: paste Page inline CSS.
  7. Create one region: Title FAQ for IT, Type Dynamic Content, Language PL/SQL, PL/SQL Function Body returning a CLOB: paste Region source.
  8. Region → Appearance → Template Blank with Attributes.
  9. Region → Advanced → Static ID afh-faq. Without it the search, Expand all and section links do nothing (the JavaScript finds the region by this ID).
  10. Save.

Page load JavaScript

JavaScript · Execute when Page Loads
(function () {
  var root = document.getElementById('afh-faq');
  if (!root) { return; }
  var input = root.querySelector('.afh-faq-input');
  var result = root.querySelector('.afh-faq-result');
  var empty = root.querySelector('p.afh-faq-empty');
  var toggle = root.querySelector('.afh-faq-toggle');
  var qas = Array.prototype.slice.call(root.querySelectorAll('.afh-faq-qa'));
  var secs = Array.prototype.slice.call(root.querySelectorAll('.afh-faq-sec'));
  if (!input || !qas.length) { return; }
  qas.forEach(function (d) { d.afhText = d.textContent.toLowerCase(); d.afhOpen = false; });

  function clearMarks(el) {
    el.querySelectorAll('mark').forEach(function (m) { m.replaceWith(document.createTextNode(m.textContent)); });
    el.normalize();
  }
  function markIn(el, term) {
    var walker = document.createTreeWalker(el, NodeFilter.SHOW_TEXT, null);
    var nodes = [];
    var n = walker.nextNode();
    while (n) { if (n.nodeValue.toLowerCase().indexOf(term) !== -1) { nodes.push(n); } n = walker.nextNode(); }
    nodes.forEach(function (node) {
      var s = node.nodeValue;
      var lower = s.toLowerCase();
      var frag = document.createDocumentFragment();
      var i = 0;
      var j = lower.indexOf(term, i);
      while (j !== -1) {
        frag.appendChild(document.createTextNode(s.slice(i, j)));
        var m = document.createElement('mark');
        m.textContent = s.slice(j, j + term.length);
        frag.appendChild(m);
        i = j + term.length;
        j = lower.indexOf(term, i);
      }
      frag.appendChild(document.createTextNode(s.slice(i)));
      node.replaceWith(frag);
    });
  }
  function run() {
    var term = input.value.trim().toLowerCase();
    var shown = 0;
    qas.forEach(function (d) {
      clearMarks(d);
      var hit = !term || d.afhText.indexOf(term) !== -1;
      d.classList.toggle('is-hidden', !hit);
      if (term && hit) { d.open = true; markIn(d, term); shown += 1; } else if (!term) { d.open = d.afhOpen; }
    });
    secs.forEach(function (s) {
      var any = !!s.querySelector('.afh-faq-qa:not(.is-hidden)');
      s.classList.toggle('is-hidden', !any);
      var link = root.querySelector('.afh-faq-toc a[href="#' + s.id + '"]');
      if (link) { link.classList.toggle('is-empty', !any); }
    });
    empty.hidden = !(term && shown === 0);
    result.textContent = term ? shown + ' of ' + qas.length + ' questions match' : '';
  }

  var timer;
  input.addEventListener('input', function () { clearTimeout(timer); timer = setTimeout(run, 120); });
  qas.forEach(function (d) {
    d.addEventListener('toggle', function () { if (!input.value.trim()) { d.afhOpen = d.open; } });
  });
  toggle.addEventListener('click', function () {
    var open = toggle.textContent === 'Expand all';
    qas.forEach(function (d) { if (!d.classList.contains('is-hidden')) { d.open = open; d.afhOpen = open; } });
    toggle.textContent = open ? 'Collapse all' : 'Expand all';
  });
  document.addEventListener('keydown', function (e) {
    var t = e.target;
    var typing = t && (t.tagName === 'INPUT' || t.tagName === 'TEXTAREA' || t.isContentEditable);
    if (e.key === '/' && !typing) { e.preventDefault(); input.focus(); }
    if (e.key === 'Escape' && t === input) { input.value = ''; run(); input.blur(); }
  });
  function openHash() {
    var id = decodeURIComponent(location.hash.slice(1));
    var el = id && document.getElementById(id);
    if (el && el.classList.contains('afh-faq-qa')) { el.open = true; el.afhOpen = true; el.scrollIntoView({ block: 'start' }); }
  }
  window.addEventListener('hashchange', openHash);
  openHash();
})();

Region source

PL/SQL · Function Body returning a CLOB
DECLARE
  l_file BLOB;
BEGIN
  SELECT file_content INTO l_file
    FROM apex_application_static_files
   WHERE application_id = :APP_ID
     AND file_name = 'afh_faq.html';
  RETURN '<div class="afh-faq-bar">'
      || '<label class="afh-faq-search"><span class="fa fa-search" aria-hidden="true"></span>'
      || '<input type="search" class="afh-faq-input" placeholder="Search questions, page numbers, error codes, job names"'
      || ' aria-label="Search the FAQ" autocomplete="off"></label>'
      || '<button type="button" class="t-Button afh-faq-toggle">Expand all</button></div>'
      || '<p class="afh-faq-result" aria-live="polite"></p>'
      || '<p class="afh-faq-empty" hidden>No question matches that search.</p>'
      || apex_util.blob_to_clob(l_file);
EXCEPTION
  WHEN NO_DATA_FOUND THEN
    RETURN '<p class="afh-faq-empty">The FAQ file afh_faq.html is missing from this application. '
        || 'Re-import the application from the release bundle.</p>';
END;

Add FAQ for IT under Administration in the navigation menu

Steps

  1. SQL Workshop → SQL Commands in the Hub's workspace, parsing schema AFH.
  2. Run Menu entry below. Expect 1 row merged. Running it again is harmless.
  3. Sign out and back in as an application owner or admin. Administration → FAQ for IT opens the page.

Menu entry

SQL · run as AFH
-- Adds "FAQ for IT" under Administration in the navigation menu. Safe to run twice.
MERGE INTO afh_nav_entries t
USING (SELECT 'ADMIN_FAQ' AS entry_code FROM dual) s
   ON (t.entry_code = s.entry_code)
 WHEN MATCHED THEN UPDATE
  SET t.list_code         = 'MENU',
      t.parent_entry_code = 'ADMIN',
      t.label             = 'FAQ for IT',
      t.target_type       = 'PAGE',
      t.page_no           = 68,
      t.icon_css          = 'fa-question-circle',
      t.display_seq       = 78,
      t.audience_code     = 'APP_OWNER_OR_ADMIN',
      t.badge_source      = NULL
 WHEN NOT MATCHED THEN INSERT
      (list_code, entry_code, parent_entry_code, label, target_type, page_no,
       icon_css, display_seq, audience_code, badge_source)
VALUES ('MENU', 'ADMIN_FAQ', 'ADMIN', 'FAQ for IT', 'PAGE', 68,
        'fa-question-circle', 78, 'APP_OWNER_OR_ADMIN', NULL);

COMMIT;
